💻 DocMindChatIframe 完整调用示例

// 1. 在其他系统页面中引入当前部署的 DocMind SDK。
<script src="http://192.168.1.220:5174/chat-iframe/docmind-chat-iframe-parent.js"></script>

// 业务上下文：用于生成当前业务的独立会话 scope。
const businessContext = {
  // 外部系统 ID，只允许字母和数字。
  source_system:      'oa',
  // 外部系统功能 ID，用于生成业务会话 scope。
  function_id:        'incomingDocument',
  // 当前业务页面 ID，用于生成业务会话 scope。
  business_id:        'incomingDetail37908',
  // 外部系统用户 ID，只允许字母和数字。
  external_user_id:   '1001',
  // 外部系统用户显示名，用于 docMind 后台识别。
  external_user_name: '张三',
}

// 来文级公共元数据：同一来文的全部附件共享；首次打开助手触发未入库附件同步时会提交这些字段。
const incomingDocumentMetadata = {
  // 当前页面中的来文 ID；它与页面级 business_id 语义不同。
  source_doc_id:    '37908',
  // 来文编号。
  document_number: '上铁辆〔2020〕316号',
  // 来文标题。
  title:           '中国铁路上海局集团有限公司关于重新印发《路用客车检修运用管理办法》的通知',
  // 来文类别。
  incoming_type:   '集团公司文件',
  // 来文单位。
  source_unit:     '安全科',
  // 来文日期，格式 YYYY-MM-DD。
  incoming_date:   '2020-10-20',
}

const chat = new DocMindChatIframe({
  // 当前 5174 Nginx 已代理 /api，因此只需填写 iframe 地址，不要填写 apiBaseUrl。
  iframeSrc: 'http://192.168.1.220:5174/chat-iframe/',
  ...businessContext,
  document_metadata: incomingDocumentMetadata,
})

// 可选参数速查（按需启用，未列出的使用默认值；详见 docmind-chat-iframe-parent.js 构造函数）：
//   iframeSrc:          'http://192.168.1.220:5174/chat-iframe/' // iframe 完整入口；SDK 也从该地址加载时可省略
//   apiBaseUrl:         当前部署无需填写                         // 默认使用 iframe origin，经 5174 的 /api 代理访问后端
//   agentId:            'default-chatbot'               // 后端智能体 ID
//   tokenExchangeUrl:   '/api/chat-iframe/token'        // 信任后端换 token 直连地址（绕过同源代理）
//   position:           'bottom-right'                  // 悬浮位置：bottom-right/left、top-right/left
//   width / height:     460 / 680                       // 普通窗口尺寸（像素）
//   initialState:       'minimized'                     // 初始状态：minimized / normal / maximized / closed
//   offsetX / offsetY:  24 / 24                         // 距视口边缘偏移（像素）
//   buttonHtml:         '🤖'               // 自定义悬浮按钮内容
//   document_metadata: { source_doc_id, ... }           // 来文 ID 与公共元数据，首次打开助手发现未入库时提交

// SPA 切换业务页面时先更新页面身份；该调用会清空上一页面的 SDK 任务与 iframe 状态。
// chat.setPageContext({ source_system: 'oa', source_function_id: 'incomingDocument', business_id: 'incomingDetail37909', document_metadata: nextIncomingDocumentMetadata })
// 随后为新页面重新调用 setPageContent() 和 setFiles()。

// 2. 设置自定义页面内容（可选，设置后不再实时获取网页HTML，且不转markdown）
chat.setPageContent([
  '页面标题：中国铁路上海局集团有限公司关于重新印发…的通知',
  '来文 ID：37908；来文类别：集团公司通知；来文单位：供电部；来文日期：2020-10-20',
  '主文件：上铁辆〔2020〕316号.pdf（428.76 KB）',
  '附件：附件4.xls',
  '附件：附件5.doc',
].join('\n'))

// 3. 设置附件列表（被助手默认带入；selected: true 的项会被高亮选中）
//   name：文件名，用于助手展示和上传时的文件名。
//   source_file_id：来源系统内该附件的稳定唯一 ID，用于匹配与关联；请使用嵌入系统的文件 ID。
//   source_url：附件下载地址；待入库时由 DocMind 后端直接下载，必须是后端可访问的 HTTP/HTTPS 地址。
//               相对地址会由父 SDK 按当前嵌入页面补成绝对地址，不要求浏览器跨域读取附件内容。
//   selected：是否在助手中默认选中。
chat.setFiles([
  {
    name:       '上铁辆〔2020〕316号.pdf',
    source_file_id: '202010200206',
    source_url: '/chat-iframe/test1/上铁辆〔2020〕316号.pdf',
    selected:   true
  },
  {
    name:       '附件4.xls',
    source_file_id: '202010200207',
    source_url: '/chat-iframe/test1/附件4.xls'
  },
  {
    name:       '附件5.doc',
    source_file_id: '202010200208',
    source_url: '/chat-iframe/test1/附件5.doc'
  },
])

// 4. 监听助手事件（可选）
chat.on('conversationCreated', (data) => console.log('新会话', data.conversationId))
chat.on('messageSent',         (data) => console.log('用户提问', data.query))
chat.on('stateChange',         (data) => console.log('窗口状态', data.state))
