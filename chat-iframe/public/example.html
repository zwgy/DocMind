💻 DocMindChatIframe 完整调用示例

// 1. 构造悬浮助手（autoInit: false 让初始化顺序在代码中显式呈现）
// 当前代码运行在宿主业务页：window.location.origin 为“协议 + 域名 + 端口”，例如 https://oa.example.com，不含路径。
const hostOrigin = window.location.origin
// 同源部署时 iframe 与宿主相同；跨域部署请改为 iframe 服务域，例如 https://docmind.example.com。
const iframeOrigin = hostOrigin

const chat = new DocMindChatIframe({
  iframeSrc: './', // iframe 入口；跨域时改为 iframeOrigin + '/chat-iframe/'
  targetOrigin: iframeOrigin, // 填 iframe 页面所在域：SDK 只向此域 postMessage，不能填 *
  originAllowlist: [hostOrigin], // 填允许嵌入 iframe 的宿主页域；多个宿主可列出多个精确域

  // 业务身份（必填，缺失会抛错）
  source_system:      'oa',
  function_id:        'incomingDocument',
  business_id:        '37908',
  external_user_id:   '1001',
  external_user_name: '张三',
})

// 可选参数速查（按需启用，未列出的使用默认值；详见 docmind-chat-iframe-parent.js 构造函数）：
//   apiBaseUrl:         'http://api.example.com'        // 后端地址；留空走同源 /api 代理
//   agentId:            'default-chatbot'               // 后端智能体 ID
//   tokenExchangeUrl:   '/api/chat-iframe/token'        // 信任后端换 token 直连地址（绕过同源代理）
//   targetOrigin:       'https://docmind.example.com'   // iframe 页面 origin，只填协议/域名/端口，不带路径，禁止 *
//   originAllowlist:    ['https://oa.example.com']      // 宿主业务页 origin；多个可信宿主逐项列出，禁止 *
//   iframe.contentWindow: SDK 创建的 iframe 内部 window；它用于校验消息来源，无需接入方填写
//   position:           'bottom-right'                  // 悬浮位置：bottom-right/left、top-right/left
//   width / height:     460 / 680                       // 普通窗口尺寸（像素）
//   initialState:       'minimized'                     // 初始状态：minimized / normal / maximized / closed
//   includePageContent: true                            // iframe 是否自动抓取页面 HTML
//   includeFiles:       true                            // iframe 是否自动拉取附件列表
//   offsetX / offsetY:  24 / 24                         // 距视口边缘偏移（像素）
//   buttonHtml:         '🤖'               // 自定义悬浮按钮内容

// 2. 设置自定义页面内容（可选，设置后不再实时获取网页HTML，且不转markdown）
chat.setPageContent([
  '页面标题：中国铁路上海局集团有限公司关于重新印发…的通知',
  '来文 ID：37908；来文类别：集团公司通知；来文单位：供电部；来文日期：2020-10-20',
  '主文件：上铁辆〔2020〕316号.pdf（428.76 KB）',
].join('\n'))

// 3. 设置附件列表（被助手默认带入；selected: true 的项会被高亮选中）
chat.setFiles([
  {
    id:         '202010200206',
    name:       '上铁辆〔2020〕316号.pdf',
    source_url: '/chat-iframe/test1/上铁辆〔2020〕316号.pdf',
    type:       'document',
    selected:   true,
  },
  {
    id:         '202010200207',
    name:       '附件4.xls',
    source_url: '/chat-iframe/test1/附件4.xls',
    type:       'document',
  },
  {
    id:         '202010200209',
    name:       '附件1～3.doc',
    source_url: '/chat-iframe/test1/附件1～3.doc',
    type:       'document',
  },
])

// 4. 监听助手事件（可选）
chat.on('conversationCreated', (data) => console.log('新会话', data.conversationId))
chat.on('messageSent',         (data) => console.log('用户提问', data.query))
chat.on('stateChange',         (data) => console.log('窗口状态', data.state))

// 5. 显式初始化（autoInit: false 时必调）
chat.init()
